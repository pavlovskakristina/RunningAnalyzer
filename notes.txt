1. DESTRUCTURING 
Destructuring Objects

const person = {
  firstName: "John",
  lastName: "Doe",
  age: 50,
  car: {
    brand: 'Ford',
    model: 'Mustang',
  }
};

// Destructuring
let {firstName, lastName, age, car: { brand, model }} = person;

let message = `My name is ${firstName}, and I drive a ${brand} ${model}.`;

//You can now access each variable separately:
document.getElementById("demo").innerHTML = firstName;


useState Hook Destructuring
import { createRoot, useState } from 'react-dom/client'

function Counter() {
  // Destructuring the array returned by useState
  const [count, setCount] = useState(0);
  
  return (
    <button onClick={() => setCount(count + 1)}>
      Count: {count}
    </button>
  );
}

createRoot(document.getElementById('root')).render(
  <Counter />
);


2. Spread Operator
const fruits = ["jabłko", "banan"];
const vegetables = ["marchew", "cebula"];

// KOPIOWANIE: Tworzymy nową tablicę i "rozsypujemy" do niej owoce
const fruitsCopy = [...fruits]; 

// DODAWANIE: Rozsypujemy owoce i dorzucamy coś nowego
const moreFruits = [...fruits, "truskawka", "kiwi"]; 
// Wynik: ["jabłko", "banan", "truskawka", "kiwi"]

// ŁĄCZENIE: Rozsypujemy zawartość obu tablic do jednej nowej
const groceries = [...fruits, ...vegetables];
// Wynik: ["jabłko", "banan", "marchew", "cebula"]
